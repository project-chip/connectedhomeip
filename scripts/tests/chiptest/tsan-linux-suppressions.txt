# The Linux server app ends up with a data race in libglib.  A race_top
# suppression does not work, since the actual race is inside a memset, so the
# thing on top of the stack is memset.  called_from_lib is a narrower
# suppression than a "race" suppression (which would be "libglib anywhere on the
# stack", as opposed to "inside a function TSan intercepts, which was called
# from libglib).
#
# See https://github.com/project-chip/connectedhomeip/issues/14710 for
# addressing this.
#
# The sanitizer unit-test step uses this file too: TestBleFallback drives a
# real GDBus peer whose worker threads live in uninstrumented GIO.
called_from_lib:libglib-2.0.so
called_from_lib:libgio-2.0.so
called_from_lib:libgobject-2.0.so
#
# GLib allocates the implementation of a GRecMutex on first use and publishes
# it with its own atomics, which TSan does not see either.
race:g_rec_mutex_impl_new
#
# g_rec_mutex_unlock tail-calls pthread_mutex_unlock, so TSan attributes the
# unlock in G_DEFINE_TYPE's *_get_type_once to the generated code instead of
# libglib and the called_from_lib entry above does not cover it.
mutex:_get_type_once$
